// ==================== (HTTP GET JNI)
static std::string FetchUrlContent(const std::string& url) {
 if (!g_JavaVM) return "";
 JNIEnv* env = nullptr;
 bool detach = false;
 int st = g_JavaVM->GetEnv((void**)&env, JNI_VERSION_1_6);
 if (st == JNI_EDETACHED) { g_JavaVM->AttachCurrentThread(&env, nullptr); detach = true; }
 else if (st != JNI_OK) return "";

 std::string result;
 jclass urlClass = env->FindClass("java/net/URL");
 if (!urlClass) { if (detach) g_JavaVM->DetachCurrentThread(); return ""; }
 jmethodID urlCtor = env->GetMethodID(urlClass, "<init>", "(Ljava/lang/String;)V");
 jstring urlStr = env->NewStringUTF(url.c_str());
 jobject urlObj = env->NewObject(urlClass, urlCtor, urlStr);
 if (urlObj) {
 jmethodID openStream = env->GetMethodID(urlClass, "openStream", "()Ljava/io/InputStream;");
 jobject stream = env->CallObjectMethod(urlObj, openStream);
 if (stream) {
 jclass isClass = env->GetObjectClass(stream);
 jmethodID readM = env->GetMethodID(isClass, "read", "([B)I");
 jbyteArray buf = env->NewByteArray(4096);
 while (true) {
 jint n = env->CallIntMethod(stream, readM, buf);
 if (n <= 0 || env->ExceptionCheck()) { env->ExceptionClear(); break; }
 jbyte* bytes = env->GetByteArrayElements(buf, nullptr);
 result.append((char*)bytes, n);
 env->ReleaseByteArrayElements(buf, bytes, JNI_ABORT);
 }
 env->DeleteLocalRef(buf);
 env->DeleteLocalRef(isClass);
 env->DeleteLocalRef(stream);
 }
 env->DeleteLocalRef(urlObj);
 }
 env->DeleteLocalRef(urlStr);
 env->DeleteLocalRef(urlClass);
 if (env->ExceptionCheck()) env->ExceptionClear();
 if (detach) g_JavaVM->DetachCurrentThread();
 return result;
}

 if (!g_JavaVM || !g_GlobalContext) return "unknown";

 bool detach = false;
 int st = g_JavaVM->GetEnv((void**)&env, JNI_VERSION_1_6);
 if (st == JNI_EDETACHED) { g_JavaVM->AttachCurrentThread(&env, nullptr); detach = true; }
 else if (st != JNI_OK) return "unknown";

 jclass secureClass = env->FindClass("android/provider/Settings$Secure");
 if (!secureClass) { if (detach) g_JavaVM->DetachCurrentThread(); return "unknown"; }
 jmethodID getString = env->GetStaticMethodID(secureClass, "getString",
 "(Landroid/content/ContentResolver;Ljava/lang/String;)Ljava/lang/String;");

 jmethodID getResolver = env->GetMethodID(ctxClass,
 "getContentResolver", "()Landroid/content/ContentResolver;");
 jobject resolver = env->CallObjectMethod(g_GlobalContext, getResolver);
 env->DeleteLocalRef(ctxClass);
 jstring name = env->NewStringUTF("android_id");
 jstring id = (jstring)env->CallStaticObjectMethod(secureClass, getString, resolver, name);
 if (id) {
 const char* s = env->GetStringUTFChars(id, nullptr);
 g_deviceHwid = s;
 env->ReleaseStringUTFChars(id, s);
 env->DeleteLocalRef(id);
 }
 env->DeleteLocalRef(name);
 env->DeleteLocalRef(resolver);
 env->DeleteLocalRef(secureClass);
 if (env->ExceptionCheck()) env->ExceptionClear();
 if (detach) g_JavaVM->DetachCurrentThread();

 std::string out;
 for (char c : g_deviceHwid) if (!isspace((unsigned char)c)) out += tolower((unsigned char)c);

bool CopyToClipboardAndroid(const std::string& text) {
 if (!g_JavaVM || !g_GlobalContext) return false;
 JNIEnv* env = nullptr;
 bool detach = false;
 int st = g_JavaVM->GetEnv((void**)&env, JNI_VERSION_1_6);
 if (st == JNI_EDETACHED) { g_JavaVM->AttachCurrentThread(&env, nullptr); detach = true; }
 else if (st != JNI_OK) return false;

 jclass ctxClass = env->GetObjectClass(g_GlobalContext);
 jstring serviceName = env->NewStringUTF("clipboard");
 jmethodID getService = env->GetMethodID(ctxClass,
 "getSystemService", "(Ljava/lang/String;)Ljava/lang/Object;");
 jobject cm = env->CallObjectMethod(g_GlobalContext, getService, serviceName);
 env->DeleteLocalRef(ctxClass);
 bool ok = false;
 if (cm) {
 jclass cmClass = env->FindClass("android/content/ClipboardManager");
 jclass clipClass = env->FindClass("android/content/ClipData");
 jmethodID newPlain = env->GetStaticMethodID(clipClass, "newPlainText",
 "(Ljava/lang/CharSequence;Ljava/lang/CharSequence;)Landroid/content/ClipData;");
 jstring label = env->NewStringUTF("hwid");
 jstring txt = env->NewStringUTF(text.c_str());
 jobject clip = env->CallStaticObjectMethod(clipClass, newPlain, label, txt);
 jmethodID setClip = env->GetMethodID(cmClass, "setPrimaryClip", "(Landroid/content/ClipData;)V");
 env->CallVoidMethod(cm, setClip, clip);
 ok = !env->ExceptionCheck();
 env->DeleteLocalRef(label); env->DeleteLocalRef(txt); env->DeleteLocalRef(clip);
 env->DeleteLocalRef(cmClass); env->DeleteLocalRef(clipClass); env->DeleteLocalRef(cm);
 }
 env->DeleteLocalRef(serviceName);
 if (env->ExceptionCheck()) env->ExceptionClear();
 if (detach) g_JavaVM->DetachCurrentThread();
 return ok;
}
